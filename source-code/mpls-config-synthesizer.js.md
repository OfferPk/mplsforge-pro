 \* MPLSForge Pro \- Multi-Vendor Configuration Synthesizer  
 \* Generates verified production syntax for Cisco IOS-XR, Juniper Junos,

 \*/

(function (root, factory) {  
  if (typeof module \=== 'object' && module.exports) {  
    module.exports \= factory();  
  } else {  
    root.MPLSConfigSynthesizer \= factory();  
  }

  /\*\*  
   \* Generates Cisco IOS-XR configuration  
   \*/  
  function generateCiscoIOSXR(cfg) {  
    const lines \= \[\];  
    lines.push('\! \=======================================================');  
    lines.push('\! Cisco IOS-XR MPLS Backbone Configuration');  
    lines.push(\`\! Generated by MPLSForge Pro for Node: \${cfg.hostname || 'PE-01'}\`);  
    lines.push('\! \=======================================================');  
    lines.push('');

    // Global MPLS & TTL propagation  
    if (cfg.ttlPropagate \=== false) {  
      lines.push('mpls ip-ttl-propagate disable');  
      lines.push('\!');  
    }

    // Core Interfaces MTU & MPLS activation  
    const iface \= cfg.coreInterface || 'HundredGigE0/0/0/0';  
    const mtu \= cfg.interfaceMtu || 9216;  
    lines.push(\`interface \${iface}\`);  
    lines.push(\` description Core-Backbone-Trunk \[MPLSForge-Engineered\]\`);  
    lines.push(\` mtu \${mtu}\`);  
    lines.push(\` ipv4 address \${cfg.coreIp || '10.100.1.1 255.255.255.252'}\`);  
    lines.push(\` mpls\`);  
    lines.push(\`  mtu \${mtu}\`);  
    lines.push(' \!');  
    lines.push('\!');

    // Transport: LDP vs SR-MPLS  
    if (cfg.transportProtocol \=== 'SR-MPLS') {  
      lines.push('router isis CORE');  
      lines.push(' is-type level-2-only');  
      lines.push(' net 49.0001.0001.0001.0001.00');  
      lines.push(' segment-routing mpls');  
      lines.push(' address-family ipv4 unicast');  
      lines.push('  metric-style wide');  
      lines.push('  segment-routing mpls');  
      lines.push(' \!');  
      lines.push(\` interface \${iface}\`);  
      lines.push('  point-to-point');  
      lines.push('  address-family ipv4 unicast');  
      if (cfg.frrMode \=== 'TI\_LFA') {  
        lines.push('   fast-reroute per-prefix');  
        lines.push('   fast-reroute per-prefix ti-lfa');  
      }  
      lines.push('  \!');  
      lines.push(' \!');  
    } else {  
      // LDP  
      lines.push('mpls ldp');  
      lines.push(\` router-id \${cfg.routerId || '10.0.0.1'}\`);  
      if (cfg.entropyLabel) {  
        lines.push(' entropy-label');  
      }  
      if (cfg.authType \=== 'MD5') {  
        lines.push(' neighbor 10.0.0.2 password encrypted 0822455D1A16');  
      } else if (cfg.authType \=== 'TCP\_AO') {  
        lines.push(' neighbor 10.0.0.2 authentication tcp-ao KEYCHAIN\_CORE');  
      }  
      if (cfg.ldpIgpSync) {  
        const hold \= cfg.ldpHoldTimerSeconds || 15;  
        lines.push(' igp sync delay on-session-up ' \+ hold);  
      }  
      lines.push(' session protection');  
      lines.push(\` interface \${iface}\`);  
      lines.push(' \!');  
      lines.push('\!');  
    }

    // Explicit Null for IPv6 6PE / 6VPE  
    if (cfg.phpMode \=== 'EXPLICIT\_NULL\_V6' || cfg.hasIpv6) {  
      lines.push('mpls ipv6 explicit-null');  
    } else if (cfg.phpMode \=== 'EXPLICIT\_NULL\_V4') {  
      lines.push('mpls ipv4 explicit-null');  
    }

    return lines.join('\\n');  
  }

  /\*\*  
   \* Generates Juniper Junos configuration  
   \*/  
  function generateJuniperJunos(cfg) {  
    const lines \= \[\];  
    lines.push('\# \=======================================================');  
    lines.push('\# Juniper Junos MPLS Backbone Configuration');  
    lines.push(\`\# Generated by MPLSForge Pro for Node: \${cfg.hostname || 'PE-01'}\`);  
    lines.push('\# \=======================================================');  
    lines.push('');

    const iface \= cfg.coreInterface || 'et-0/0/0';  
    const mtu \= cfg.interfaceMtu || 9216;

    // Interface MTU & family mpls  
    lines.push(\`set interfaces \${iface} description "Core-Backbone-Trunk"\`);  
    lines.push(\`set interfaces \${iface} mtu \${mtu}\`);  
    lines.push(\`set interfaces \${iface} unit 0 family inet address \${cfg.coreIp || '10.100.1.1/30'}\`);  
    lines.push(\`set interfaces \${iface} unit 0 family mpls mtu \${mtu}\`);  
    lines.push('');

    // Protocols MPLS  
    lines.push(\`set protocols mpls interface \${iface}.0\`);  
    if (cfg.ttlPropagate \=== false) {  
      lines.push('set protocols mpls no-propagate-ttl');  
    }  
    if (cfg.phpMode \=== 'EXPLICIT\_NULL\_V6' || cfg.hasIpv6) {  
      lines.push('set protocols mpls ipv6-tunneling');  
      lines.push('set protocols mpls explicit-null');  
    } else if (cfg.phpMode \=== 'EXPLICIT\_NULL\_V4') {  
      lines.push('set protocols mpls explicit-null');  
    }  
    if (cfg.entropyLabel) {  
      lines.push('set protocols mpls entropy-label');  
    }  
    lines.push('');

    // Protocols LDP  
    if (cfg.transportProtocol \!== 'SR-MPLS') {  
      lines.push(\`set protocols ldp interface \${iface}.0\`);  
      if (cfg.ldpIgpSync) {  
        const hold \= cfg.ldpHoldTimerSeconds || 15;  
        lines.push(\`set protocols ldp ldp-synchronization hold-time \${hold}\`);  
      }  
      lines.push('set protocols ldp session-protection');  
      if (cfg.authType \=== 'MD5') {  
        lines.push('set protocols ldp md5-key "\$9\$JzG8X7NDk."');  
      }  
    } else {  
      lines.push('set protocols isis source-packet-routing srgb start-label 16000 index-range 8000');  
      lines.push(\`set protocols isis interface \${iface}.0 level 2 post-convergence-lfa\`);  
    }

    return lines.join('\\n');  
  }

  /\*\*  
   \* Generates Nokia SR-OS (TiMOS) configuration  
   \*/  
  function generateNokiaSROS(cfg) {  
    const lines \= \[\];  
    lines.push('\# \=======================================================');  
    lines.push('\# Nokia SR-OS MPLS Core Configuration');  
    lines.push(\`\# Generated by MPLSForge Pro for Node: \${cfg.hostname || 'PE-01'}\`);  
    lines.push('\# \=======================================================');  
    lines.push('');

    const iface \= cfg.coreInterface || '1/1/c1/1';  
    const mtu \= cfg.interfaceMtu || 9216;

    lines.push(\`/configure port \${iface} ethernet mtu \${mtu}\`);  
    lines.push(\`/configure port \${iface} no shutdown\`);  
    lines.push('');  
    lines.push('/configure router Base');  
    lines.push('    interface "to-core-p1"');  
    lines.push(\`        address \${cfg.coreIp || '10.100.1.1/30'}\`);  
    lines.push(\`        port \${iface}\`);  
    lines.push(\`        mpls-mtu \${mtu}\`);  
    lines.push('        no shutdown');  
    lines.push('    exit');  
    lines.push('');

    lines.push('    mpls');  
    if (cfg.ttlPropagate \=== false) {  
      lines.push('        ttl-propagate');  
      lines.push('            local false');  
      lines.push('            transit false');  
      lines.push('        exit');  
    }  
    if (cfg.entropyLabel) {  
      lines.push('        entropy-label');  
    }  
    lines.push('        interface "to-core-p1"');  
    lines.push('            no shutdown');  
    lines.push('        exit');  
    lines.push('        no shutdown');  
    lines.push('    exit');  
    lines.push('');

    if (cfg.transportProtocol \!== 'SR-MPLS') {  
      lines.push('    ldp');  
      if (cfg.ldpIgpSync) {  
        const hold \= cfg.ldpHoldTimerSeconds || 15;  
        lines.push(\`        ldp-sync-timers hold-down \${hold}\`);  
      }  
      if (cfg.authType \=== 'MD5') {  
        lines.push('        interface-parameters');  
        lines.push('            interface "to-core-p1"');  
        lines.push('                auth-key "CoreSecretKey123"');  
        lines.push('            exit');  
        lines.push('        exit');  
      }  
      lines.push('        interface "to-core-p1"');  
      lines.push('            no shutdown');  
      lines.push('        exit');  
      lines.push('        no shutdown');  
      lines.push('    exit');  
    }

    return lines.join('\\n');  
  }

  /\*\*  
   \* Generates Linux Kernel MPLS configuration  
   \*/  
  function generateLinuxKernel(cfg) {  
    const lines \= \[\];  
    lines.push('\#\!/usr/bin/env bash');  
    lines.push('\# \=======================================================');  
    lines.push('\# Linux Kernel MPLS Data Plane Initialization');  
    lines.push(\`\# Generated by MPLSForge Pro for Node: \${cfg.hostname || 'PE-01'}\`);  
    lines.push('\# \=======================================================');  
    lines.push('set \-euo pipefail');  
    lines.push('');  
    lines.push('\# 1\. Load Linux MPLS Kernel Modules');  
    lines.push('modprobe mpls\_router');  
    lines.push('modprobe mpls\_iptunnel');  
    lines.push('modprobe mpls\_gso');  
    lines.push('');  
    lines.push('\# 2\. Configure Sysctl MPLS Label Space & Interfaces');  
    lines.push('sysctl \-w net.mpls.platform\_labels=1048575');  
    const iface \= cfg.coreInterface || 'eth0';  
    lines.push(\`sysctl \-w net.mpls.conf.\${iface}.input=1\`);  
    if (cfg.ttlPropagate \=== false) {  
      lines.push(\`sysctl \-w net.mpls.conf.\${iface}.ttl\_propagate=0\`);  
    }  
    lines.push('');  
    lines.push('\# 3\. Set Hardware Interface MTU to Prevent Blackholes');  
    const mtu \= cfg.interfaceMtu || 9216;  
    lines.push(\`ip link set dev \${iface} mtu \${mtu}\`);  
    lines.push(\`ip link set dev \${iface} up\`);  
    lines.push('');  
    lines.push('\# 4\. Example: Ingress Encapsulation (Push Label 100, TC 0, TTL 255)');  
    lines.push(\`ip route add 198.51.100.0/24 encap mpls 100 via inet 10.100.1.2 dev \${iface}\`);  
    lines.push('');  
    lines.push('\# 5\. Example: Transit LSR Swap (Swap Ingress 100 \-\> Egress 200)');  
    lines.push(\`ip \-f mpls route add 100 as to 200 via inet 10.100.1.2 dev \${iface}\`);  
    lines.push('');  
    lines.push('\# 6\. Example: Egress LER Pop (Penultimate Hop Popping / De-encapsulation)');  
    lines.push(\`ip \-f mpls route add 200 dev lo\`);  
    lines.push('echo "\[OK\] Linux Kernel MPLS data plane configured successfully."');

    return lines.join('\\n');  
  }

  /\*\*  
   \* Generates Prometheus MPLS Metrics Exporter Python Script  
   \*/  
  function generatePrometheusExporter(cfg) {  
    const lines \= \[\];  
    lines.push('\#\!/usr/bin/env python3');  
    lines.push('"""');  
    lines.push('MPLSForge Pro \- Prometheus Core MPLS Telemetry Exporter');  
    lines.push('Exposes LDP session states, MTU headroom, and FRR status on :9129/metrics');  
    lines.push('"""');  
    lines.push('from http.server import HTTPServer, BaseHTTPRequestHandler');  
    lines.push('import time');  
    lines.push('');  
    lines.push('CONFIG \= {');  
    lines.push(\`    "hostname": "\${cfg.hostname || 'PE-01'}",\`);  
    lines.push(\`    "interface": "\${cfg.coreInterface || 'HundredGigE0/0/0/0'}",\`);  
    lines.push(\`    "interface\_mtu": \${cfg.interfaceMtu || 9216},\`);  
    lines.push(\`    "required\_mpls\_mtu": \${(cfg.payloadMtu || 1500\) \+ ((cfg.labelCount || 2\) \* 4)},\`);  
    lines.push(\`    "ldp\_sync\_active": 1 if \${\!\!cfg.ldpIgpSync} else 0,\`);  
    lines.push(\`    "frr\_enabled": 1 if "\${cfg.frrMode || 'NONE'}" \!= "NONE" else 0,\`);  
    lines.push(\`    "entropy\_label": 1 if \${\!\!cfg.entropyLabel} else 0\`);  
    lines.push('}');  
    lines.push('');  
    lines.push('class MPLSMetricsHandler(BaseHTTPRequestHandler):');  
    lines.push('    def do\_GET(self):');  
    lines.push('        if self.path \!= "/metrics":');  
    lines.push('            self.send\_response(404)');  
    lines.push('            self.end\_headers()');  
    lines.push('            return');  
    lines.push('        ');  
    lines.push('        headroom \= CONFIG\["interface\_mtu"\] \- CONFIG\["required\_mpls\_mtu"\]');  
    lines.push('        metrics \= \[');  
    lines.push('            "\# HELP mpls\_interface\_mtu\_bytes Configured physical MTU on backbone trunk",');  
    lines.push('            "\# TYPE mpls\_interface\_mtu\_bytes gauge",');  
    lines.push('            f\\'mpls\_interface\_mtu\_bytes{{interface="{CONFIG\["interface"\]}"}} {CONFIG\["interface\_mtu"\]}\\',');  
    lines.push('            "\# HELP mpls\_mtu\_headroom\_bytes Available headroom before silent packet drop",');  
    lines.push('            "\# TYPE mpls\_mtu\_headroom\_bytes gauge",');  
    lines.push('            f\\'mpls\_mtu\_headroom\_bytes{{interface="{CONFIG\["interface"\]}"}} {headroom}\\',');  
    lines.push('            "\# HELP mpls\_ldp\_igp\_sync\_status Status of RFC 5443 LDP-IGP synchronization",');  
    lines.push('            "\# TYPE mpls\_ldp\_igp\_sync\_status gauge",');  
    lines.push('            f\\'mpls\_ldp\_igp\_sync\_status{{interface="{CONFIG\["interface"\]}"}} {CONFIG\["ldp\_sync\_active"\]}\\',');  
    lines.push('            "\# HELP mpls\_fast\_reroute\_active Status of Fast Reroute sub-50ms protection",');  
    lines.push('            "\# TYPE mpls\_fast\_reroute\_active gauge",');  
    lines.push('            f\\'mpls\_fast\_reroute\_active{{interface="{CONFIG\["interface"\]}"}} {CONFIG\["frr\_enabled"\]}\\',');  
    lines.push('            "\# HELP mpls\_entropy\_label\_enabled Status of RFC 6790 Entropy Label hashing",');  
    lines.push('            "\# TYPE mpls\_entropy\_label\_enabled gauge",');  
    lines.push('            f\\'mpls\_entropy\_label\_enabled{{interface="{CONFIG\["interface"\]}"}} {CONFIG\["entropy\_label"\]}\\'');  
    lines.push('        \]');  
    lines.push('        ');  
    lines.push('        payload \= "\\\\n".join(metrics) \+ "\\\\n"');  
    lines.push('        self.send\_response(200)');  
    lines.push('        self.send\_header("Content-Type", "text/plain; version=0.0.4")');  
    lines.push('        self.end\_headers()');  
    lines.push('        self.wfile.write(payload.encode("utf-8"))');  
    lines.push('    ');  
    lines.push('    def log\_message(self, format, \*args):');  
    lines.push('        pass');  
    lines.push('');  
    lines.push('if \_\_name\_\_ \== "\_\_main\_\_":');  
    lines.push('    server \= HTTPServer(("0.0.0.0", 9129), MPLSMetricsHandler)');  
    lines.push('    print("\[INFO\] MPLS Telemetry Exporter listening on :9129/metrics")');  
    lines.push('    try:');  
    lines.push('        server.serve\_forever()');  
    lines.push('    except KeyboardInterrupt:');  
    lines.push('        pass');

    return lines.join('\\n');  
  }
